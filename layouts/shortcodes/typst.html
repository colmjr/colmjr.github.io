{{- /* Inline a precompiled typst SVG. The body is hashed with SHA-1 and must have been compiled by scripts/render_typst.py into assets/typst-gen/. Usage: {{< typst >}}$ mat(1, 2; 3, 4) ${{< /typst >}} {{< typst inline=true >}}$ x^2 ${{< /typst >}} */ -}} {{- $src := trim .Inner " \t\r\n" -}} {{- $hash := sha1 $src -}} {{- $inline := .Get "inline" -}} {{- if not (.Page.Store.Get "typst-css") -}} {{- .Page.Store.Set "typst-css" true -}} {{- end -}} {{- $svg := resources.Get (printf "typst-gen/%s.svg" $hash) -}} {{- with $svg -}} {{ .Content | safeHTML }} {{- else -}} {{- warnf "typst SVG %s missing; run scripts/render_typst.py" $hash -}} typst not rendered, run scripts/render_typst.py {{- end -}}
